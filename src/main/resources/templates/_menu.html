/* WebSocket: подписка на обновления в реальном времени */ let stompClient = null; let reconnectTimeout = null; // Время (мс), до которого игнорировать свои собственные уведомления (~6 сек). // sessionStorage нужен, чтобы метка пережила редирект после сохранения формы. let ownChangeUntil = 0; function markOwnChange() { ownChangeUntil = Date.now() + 6000; sessionStorage.setItem('wsOwnChange', String(ownChangeUntil)); } function isOwnChange() { if (Date.now() < ownChangeUntil) { return true; } const stored = sessionStorage.getItem('wsOwnChange'); if (stored) { const ts = parseInt(stored, 10); if (!isNaN(ts) && Date.now() < ts) { ownChangeUntil = ts; return true; } sessionStorage.removeItem('wsOwnChange'); } return false; } function connectWebSocket() { try { // Нативный WebSocket (без SockJS-обёртки) — убирает несовпадение версий // SockJS-клиента и серверного iframe-транспорта (1.6.1 vs 1.0.0). const socket = new WebSocket('ws://' + window.location.host + '/ws'); stompClient = Stomp.over(socket); stompClient.debug = null; // отключаем шум в консоли stompClient.connect({}, function () { console.log('[WS] подключено к /topic/creatures'); stompClient.subscribe('/topic/creatures', function () { console.log('[WS] получено обновление'); if (isOwnChange()) { console.log('[WS] своя команда, перезагрузка пропущена'); return; } console.log('[WS] команда от другого клиента, перезагрузка страницы'); window.location.reload(); }); }, function (error) { console.error('[WS] ошибка соединения:', error); scheduleReconnect(); } ); stompClient.onclose = function () { console.warn('[WS] соединение закрыто'); scheduleReconnect(); }; } catch (e) { console.error('[WS] не удалось создать соединение:', e); scheduleReconnect(); } } function scheduleReconnect() { if (reconnectTimeout) return; reconnectTimeout = setTimeout(function () { reconnectTimeout = null; connectWebSocket(); }, 3000); } // Пометить свои действия, чтобы не прерывать свой редирект перезагрузкой. document.querySelectorAll('form').forEach(function (form) { form.addEventListener('submit', markOwnChange); }); document.querySelectorAll('a.btn-danger').forEach(function (link) { link.addEventListener('click', markOwnChange); }); connectWebSocket();
